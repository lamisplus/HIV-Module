# `enrollment_commencement` — Single Merged Table

Replaces the need to write into both `hiv_enrollment` and `hiv_art_clinical`
for the ART Enrollment & Commencement form. One record per patient enrollment.

---

## CREATE TABLE

```sql
CREATE TABLE enrollment_commencement (

    -- ── Infrastructure ────────────────────────────────────────────────────
    id                                   BIGINT          NOT NULL GENERATED BY DEFAULT AS IDENTITY,
    uuid                                 VARCHAR(255)    NOT NULL,
    unique_id_no                         VARCHAR(100),
    facility_id                          BIGINT,
    person_uuid                          VARCHAR(255)    NOT NULL,
    visit_id                             VARCHAR(255)    NOT NULL,
    vital_sign_uuid                      VARCHAR(255),
    is_commencement                      BOOLEAN         DEFAULT TRUE,
    art_status_id                        BIGINT,
    status_at_registration_id            BIGINT,
    source                               VARCHAR(255),
    latitude                             VARCHAR(255),
    longitude                            VARCHAR(255),
    archived                             INTEGER         DEFAULT 0,
    created_by                           VARCHAR(255),
    created_date                         TIMESTAMP,
    last_modified_by                     VARCHAR(255),
    last_modified_date                   TIMESTAMP,

    -- ── Registration — HIV Care & Identification ──────────────────────────
    date_enrolled_in_hiv_care            DATE            NOT NULL,
    date_confirmed_hiv_test              DATE,
    hiv_test_location                    VARCHAR(255),
    mode_of_hiv_test                     VARCHAR(50),
    care_entry_point_id                  BIGINT,
    care_entry_point_other               VARCHAR(255),
    mother_unique_id                     VARCHAR(100),

    -- ── Registration — Demographics ───────────────────────────────────────
    occupation                           VARCHAR(255),
    marital_status                       VARCHAR(50),
    educational_level                    VARCHAR(100),

    -- ── Registration — Next of Kin ────────────────────────────────────────
    nok_name                             VARCHAR(255),
    nok_relationship                     VARCHAR(100),
    nok_telephone                        VARCHAR(50),

    -- ── Registration — Prior ART & Key Population ─────────────────────────
    prior_art_code                       VARCHAR(10),
    is_kp                                BOOLEAN         DEFAULT FALSE,
    kp_typology                          VARCHAR(100),

    -- ── Registration — Transfer Details ──────────────────────────────────
    date_transferred_in                  DATE,
    facility_transferred_from            VARCHAR(255),

    -- ── Commencement — Clinical Status at ART Start ───────────────────────
    clinical_stage_id                    BIGINT,
    cd4_at_art_start                     BIGINT,
    cd4_lf                               VARCHAR(10),

    -- ── Commencement — ART Dates & Regimen ───────────────────────────────
    date_adherence_counseling_completed  DATE,
    date_art_started                     DATE            NOT NULL,
    regimen_id                           BIGINT,

    -- ── Commencement — Vitals ────────────────────────────────────────────
    weight_kg                            DOUBLE PRECISION,
    height_cm                            DOUBLE PRECISION,
    bmi                                  DOUBLE PRECISION,
    muac                                 DOUBLE PRECISION,
    muac_indication                      VARCHAR(50),
    pregnancy_status                     VARCHAR(50),

    -- ── Commencement — TB Preventive Therapy ─────────────────────────────
    tpt_medication                       VARCHAR(255),
    tpt_code                             VARCHAR(20),
    tpt_dose                             VARCHAR(100),
    tpt_start_date                       DATE,
    tpt_completion_date                  DATE,

    -- ── Constraints ──────────────────────────────────────────────────────
    CONSTRAINT pk_enrollment_commencement      PRIMARY KEY (id),
    CONSTRAINT uc_enrollment_commencement_uuid UNIQUE (uuid),

    CONSTRAINT fk_ec_person_uuid
        FOREIGN KEY (person_uuid)       REFERENCES patient_person(uuid),

    CONSTRAINT fk_ec_visit
        FOREIGN KEY (visit_id)          REFERENCES patient_visit(uuid),

    CONSTRAINT fk_ec_vital_sign
        FOREIGN KEY (vital_sign_uuid)   REFERENCES triage_vital_sign(uuid)
);
```

## Indexes

```sql
-- Primary lookup: patient + active records
CREATE INDEX idx_ec_person_uuid
    ON enrollment_commencement (person_uuid, archived);

-- Lookup by unique ART ID number
CREATE INDEX idx_ec_unique_id_no
    ON enrollment_commencement (unique_id_no)
    WHERE unique_id_no IS NOT NULL;

-- Mother linkage for PMTCT/infant queries
CREATE INDEX idx_ec_mother_unique_id
    ON enrollment_commencement (mother_unique_id)
    WHERE mother_unique_id IS NOT NULL;

-- Commencement-only queries (mirrors the old filtered index)
CREATE INDEX idx_ec_commencement_active
    ON enrollment_commencement (person_uuid)
    WHERE archived = 0 AND is_commencement = TRUE;
```

## Migration — Populate from existing tables

```sql
INSERT INTO enrollment_commencement (
    -- Infrastructure
    uuid,
    unique_id_no,
    facility_id,
    person_uuid,
    visit_id,
    vital_sign_uuid,
    is_commencement,
    art_status_id,
    status_at_registration_id,
    source,
    latitude,
    longitude,
    archived,
    created_by,
    created_date,
    last_modified_by,
    last_modified_date,

    -- Registration
    date_enrolled_in_hiv_care,
    date_confirmed_hiv_test,
    care_entry_point_id,
    care_entry_point_other,
    status_at_registration_id,
    source,

    -- Commencement
    clinical_stage_id,
    cd4_at_art_start,
    date_art_started,
    regimen_id,
    pregnancy_status,
    archived
)
SELECT
    -- Infrastructure
    e.uuid,
    e.unique_id                             AS unique_id_no,
    e.facility_id,
    e.person_uuid,
    e.visit_id,
    c.vital_sign_uuid,
    TRUE                                    AS is_commencement,
    c.art_status_id,
    e.status_at_registration_id,
    e.source,
    e.latitude,
    e.longitude,
    e.archived,
    e.created_by,
    e.created_date,
    e.last_modified_by,
    e.last_modified_date,

    -- Registration side
    e.date_of_registration                  AS date_enrolled_in_hiv_care,
    e.date_confirmed_hiv                    AS date_confirmed_hiv_test,
    e.entry_point_id                        AS care_entry_point_id,
    e.care_entry_point_other,
    e.status_at_registration_id,
    e.source,

    -- Commencement side
    c.clinical_stage_id,
    c.cd_4                                  AS cd4_at_art_start,
    c.visit_date                            AS date_art_started,
    c.regimen_id,
    c.pregnancy_status,
    c.archived

FROM hiv_enrollment e
JOIN hiv_art_clinical c
    ON  c.hiv_enrollment_uuid = e.uuid
    AND c.is_commencement     = TRUE
    AND c.archived            = 0
WHERE e.archived = 0;
```

> **Note:** Fields that did not exist in the old tables (`mode_of_hiv_test`, `hiv_test_location`,
> `mother_unique_id`, `occupation`, `marital_status`, `educational_level`, `nok_*`, `prior_art_code`,
> `is_kp`, `kp_typology`, `transfer_*`, `cd4_lf`, `date_adherence_counseling_completed`,
> `bmi`, `muac`, `muac_indication`, `tpt_*`) will be `NULL` for all migrated records.
> They are populated only on new saves through this form.
